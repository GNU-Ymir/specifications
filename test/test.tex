\section*{Twenty-two lines, four times the natural height}

\begin{brokenlisting}
import std::io;

class Point {
    let mut _x : i32;
    let mut _y : i32;

    pub self (x : i32, y : i32)
        with _x = x, _y = y
    {}

    pub fn norm (self)-> f64 {
        cast!f64 (self._x * self._x + self._y * self._y)
    }
}

fn main ()
    throws &AssertError
{
    let p = Point::new (3, 4);
    assert (p.norm () == 25.0);
    println ("norm = ", p.norm ());
}
\end{brokenlisting}
